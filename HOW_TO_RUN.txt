═══════════════════════════════════════════════════════════════

ШАГ 1: СКАЧАЙТЕ GODOT
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. Откройте: https://godotengine.org/download
2. Скачайте "Godot Engine 4.x" (Standard version)
3. Распакуйте в любую папку (например, D:\Games\Godot)
4. Запустите godot.exe

ШАГ 2: ОТКРОЙТЕ ПРОЕКТ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. В Godot нажмите кнопку "Import"
2. Перейдите в: D:\Work\UnityProjects\Allodshome_Godot
3. Нажмите "Import & Edit"
4. Подождите 5-10 секунд (импорт ассетов)

ШАГ 3: ЗАПУСТИТЕ ИГРУ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. В FileSystem (внизу) откройте: scenes → main.tscn
2. Нажмите F5 или кнопку ▶️ вверху справа
3. Игра запустится!

УПРАВЛЕНИЕ:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
• ЛКМ по земле     → Движение
• ЛКМ по врагу     → Атака
• Клавиши 1, 2, 3  → Заклинания
• Пробел           → Пауза

ЧТО ВИДЕТЬ:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
✅ Изометрический остров с зелёной травой
✅ Ваш персонаж (воин) в центре
✅ 2 орка и 1 слизень вокруг
✅ HP/Mana бары вверху слева
✅ Панель заклинаний внизу

ЕСЛИ НЕ РАБОТАЕТ:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
❌ Godot 3.x вместо 4.x → Скачайте Godot 4.4+
❌ Нет спрайтов        → Проверьте папку assets/sprites/
❌ Ошибки в Output     → Откройте панель Output (внизу)
